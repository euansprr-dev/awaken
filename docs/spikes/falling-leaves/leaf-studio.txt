# Leaf studio (docs/modules/lands.md "Leaves in the air"): a grove built in the sky at y 200 (grass floor, a 25x25
# canopy of persistent leaves on five trunks) so the leaves in the air read against a clean background. The real wind,
# a forced gust, the settle, the Elderwood's gold (its air rule forced over birch leaves) and Lumen motes at night.
# Run: PACK=perf tools/autopilot.sh docs/spikes/falling-leaves/leaf-studio.txt 1500 ; sheets: motion_sheet.py
world leafstudio creative 777
waitworld
gui 0
cmd gamerule send_command_feedback false
cmd gamerule awaken:fauna_spawning false
cmd gamerule spawn_mobs false
cmd gamerule advance_time false
cmd weather clear 1000000
cmd time set 6000
cmd awaken admin awaken
wait 200
cmd awaken powers starter 1
wait 20
screen close
window 1440 900
info
cmd awaken terrain waving status
cmd gamemode spectator
cmd tp @s 0.5 230 0.5
wait 40
cmd fill -20 200 -20 20 200 20 minecraft:grass_block
cmd fill -12 208 -12 12 209 12 minecraft:oak_leaves[persistent=true]
cmd fill 0 201 0 0 207 0 minecraft:oak_log
cmd fill -8 201 -8 -8 207 -8 minecraft:oak_log
cmd fill 8 201 -8 8 207 -8 minecraft:oak_log
cmd fill -8 201 8 -8 207 8 minecraft:oak_log
cmd fill 8 201 8 8 207 8 minecraft:oak_log
cmd tp @s 0.5 202.6 19.5 180 -6
wait 300
cmd awaken terrain wind status
cmd awaken terrain leaves status
# 1 the real wind: four frames ~13 ticks apart
shot studio_real_1
shot studio_real_2
shot studio_real_3
shot studio_real_4
cmd awaken terrain wind status
# 2 a forced gust (showcase aid)
cmd awaken terrain wind force 0.85 0.9
wait 160
shot studio_gust_1
shot studio_gust_2
shot studio_gust_3
shot studio_gust_4
cmd awaken terrain leaves status
# 3 the settle: looking down at the floor under the canopy, light wind, frames 15 ticks apart
cmd awaken terrain wind force 0.3 0.0
cmd tp @s 0.5 204.5 9.5 180 65
wait 300
shot studio_settle_1
wait 15
shot studio_settle_2
wait 15
shot studio_settle_3
wait 15
shot studio_settle_4
cmd awaken terrain wind real
# 4 the Elderwood's air (gold) over birch leaves, as the placeholder biome has them
cmd fill -12 208 -12 12 209 12 minecraft:birch_leaves[persistent=true]
cmd awaken terrain leaves biome awaken:elderwood
cmd tp @s 0.5 202.6 19.5 180 -6
wait 300
cmd awaken terrain leaves status
shot studio_elder_1
shot studio_elder_2
shot studio_elder_3
shot studio_elder_4
cmd tp @s 0.5 204.5 9.5 180 65
wait 200
shot studio_elder_floor_1
wait 15
shot studio_elder_floor_2
# 5 Lumen motes at night
cmd time set 18000
cmd awaken terrain leaves biome awaken:lumen_forest
cmd tp @s 0.5 203 22.5 180 8
wait 300
cmd awaken terrain leaves status
shot studio_motes_1
shot studio_motes_2
shot studio_motes_3
shot studio_motes_4
cmd awaken terrain leaves biome clear
cmd time set 6000
quit
