# One run, one camera per scene, waving on then off then on then off: four stills ~13 ticks apart per scene for the
# motion sheet (motion_sheet.py), and frame times for both scenes in each state (perf lines in the log).
world wave777 creative 777
waitworld
gui 0
cmd gamerule send_command_feedback false
cmd gamerule awaken:fauna_spawning false
cmd gamerule spawn_mobs false
cmd gamerule advance_time false
cmd weather clear 1000000
cmd time set 6000
cmd awaken admin awaken
wait 200
cmd awaken powers starter 1
wait 20
screen close
window 1440 900
info
cmd awaken terrain waving status
cmd gamemode spectator
cmd awaken fauna goto duskwood
wait 500
# --- waving on
look 30 -12
wait 60
shot on_ground_1
shot on_ground_2
shot on_ground_3
shot on_ground_4
cmd tp @s ~ ~14 ~
look 30 35
wait 120
shot on_canopy_1
shot on_canopy_2
shot on_canopy_3
shot on_canopy_4
perf canopy_on 20
cmd tp @s ~ ~-14 ~
look 30 -12
wait 80
perf ground_on 20
# --- waving off (resource reload: Sodium's own shader, unmarked models)
cmd awaken terrain waving off
wait 500
cmd awaken terrain waving status
shot off_ground_1
shot off_ground_2
shot off_ground_3
shot off_ground_4
cmd tp @s ~ ~14 ~
look 30 35
wait 120
shot off_canopy_1
shot off_canopy_2
shot off_canopy_3
shot off_canopy_4
perf canopy_off 20
cmd tp @s ~ ~-14 ~
look 30 -12
wait 80
perf ground_off 20
# --- on again, off again: repeat the measurement
cmd awaken terrain waving on
wait 500
cmd awaken terrain waving status
perf ground_on2 20
cmd tp @s ~ ~14 ~
look 30 35
wait 120
perf canopy_on2 20
cmd awaken terrain waving off
wait 500
perf canopy_off2 20
cmd tp @s ~ ~-14 ~
look 30 -12
wait 80
perf ground_off2 20
cmd awaken terrain waving on
wait 300
quit
