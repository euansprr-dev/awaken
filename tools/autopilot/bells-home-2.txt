# Bells stage 2 (SPEC §43.16): bells you carry and hang. A home 320 east of the city square (/awaken admin kin yard);
# the Lampwing's old bell in the nearest tree at night (glowing, asleep) and by day once beasts live there; the Ember
# Bell's card and the bell in hand; three bells on the frame at dusk; twelve, with Bell Beams 9 up and a Great Beam
# 13 up for the Fourth; the dusk song swinging them.
# ci knobs: PACK=perf,world TECTONIC=awakened
#   tools/tour-ci.sh tools/autopilot/bells-home-2.txt 2400 PACK=perf,world TECTONIC=awakened
world autopilot_bells_home_2 creative 2112026
waitworld
camera first
cmd gamerule send_command_feedback false
cmd weather clear 100000
cmd gamerule awaken:fauna_spawning false
cmd gamerule spawn_mobs false
wait 40
cmd awaken opening skip
wait 160
screen close
cmd awaken admin awaken
wait 200
screen close
cmd awaken powers starter 1
wait 20
screen close
cmd effect give @s minecraft:resistance 100000 4 true
wait 40
key dismiss
wait 10
screen close
cmd gamerule advance_time false
cmd awaken village trouble quiet on
# a fresh world holds the notice director for its first five minutes: the readings need it free
cmd tick sprint 6200
wait 200
window 1440 900
info
camera first
hud off
cmd time set 3500
cmd clear @s

# ---- 1. a home near the city, and the Lampwing's old bell in its tree ----
cmd awaken city view square
wait 300
cmd time set 6000
cmd spreadplayers ~320 ~40 0 20 false @s
wait 300
cmd summon minecraft:marker ~ ~ ~ {Tags:["yard"]}
cmd awaken admin kin yard
wait 140
screen close
wait 20
screen close
cmd gamemode spectator
cmd time set 18000
cmd awaken admin bells roost
wait 80
shot b2_01_roost_asleep_night
cmd awaken admin kin home burrowbun
cmd awaken admin kin home pebblepup
cmd tick sprint 400
cmd time set 6000
wait 120
cmd awaken admin bells roost
wait 60
shot b2_02_roost_day

# ---- 2. a bell is a thing: the card, and the bell in hand ----
cmd gamemode creative
cmd execute at @e[type=marker,tag=yard,limit=1] run tp @s ~0.5 ~ ~2.5 180 10
wait 20
hud on
cmd awaken admin bells give ember
wait 30
shotnow b2_03_bell_card
wait 200
cmd item replace entity @s weapon.mainhand with awaken:bell[awaken:bell=7]
wait 30
shotnow b2_04_bell_in_hand
hud off

# ---- 3. three on the frame, at dusk ----
cmd awaken admin bells tower 3
wait 40
cmd gamemode spectator
cmd time set 11800
cmd execute at @e[type=marker,tag=yard,limit=1] run tp @s ~0.5 ~0.4 ~-0.5 180 14
wait 60
shot b2_05_frame_three

# ---- 4. twelve: Bell Beams 9 up, a Great Beam 13 up for the Fourth ----
cmd gamemode creative
cmd time set 6000
cmd execute at @e[type=marker,tag=yard,limit=1] run tp @s ~0.5 ~ ~2.5 180 10
cmd awaken admin bells tower 12
wait 60
cmd gamemode spectator
cmd execute at @e[type=marker,tag=yard,limit=1] run tp @s ~0.5 ~7 ~11 180 8
wait 80
shot b2_06_beams_wide
cmd execute at @e[type=marker,tag=yard,limit=1] run tp @s ~0.5 ~9 ~3 180 0
wait 40
shot b2_07_beams_near

# ---- 5. the dusk song swings them ----
cmd time set 11500
cmd execute at @e[type=marker,tag=yard,limit=1] run tp @s ~0.5 ~0.4 ~-0.5 180 14
cmd awaken admin bells song
wait 30
shot b2_08_song_swing
quit
