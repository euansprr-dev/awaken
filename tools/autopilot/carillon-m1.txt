#      /Volumes/MEDIA/Awaken-carillon/tools/carillon/slot.sh light tools/autopilot.sh tools/autopilot/carillon-m1.txt 900
# A new world with Carillon on (the built-in pack, default for new worlds): the city pregenerates behind the loading
# screen ('[city] pregen done' in the log), spawn is on the Runners' Roost. Then: the sky at dawn within the S2 camera
# limits (over the north-west half, across the spire), street level in every district, the Roost, Choir Square and the
# wall walk, and the fixed studio framings of the landmarks (the same names at every milestone, so shots compare).
world carillonM1 creative 20261024
waitworld
gui 0
cmd gamerule send_command_feedback false
cmd gamerule awaken:fauna_spawning false
cmd gamerule spawn_mobs false
cmd gamerule advance_time false
cmd weather clear 1000000

cmd awaken admin awaken
wait 200
cmd awaken powers starter 1
wait 20

cmd awaken city info

look -45 12

cmd gamemode spectator
