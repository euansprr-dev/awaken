# Live enchanting (SPEC §45): the enchant-table study, the table open, then a long hold so a person (or computer use)
# can trace with a real mouse, touch the table again, and run commands from chat. On the Mac only, under slot.sh light:
#   SLOT_WHO=enchanting /Volumes/MEDIA/Awaken-carillon/tools/carillon/slot.sh light tools/autopilot.sh tools/autopilot/enchant-live.txt 1500
world autopilot_enchant_live creative 20261102
waitworld
camera first
cmd gamerule send_command_feedback false
cmd weather clear 100000
cmd gamerule awaken:fauna_spawning false
cmd gamerule spawn_mobs false
wait 40
cmd awaken opening skip
wait 160
screen close
cmd awaken admin awaken
wait 200
screen close
cmd awaken powers starter 1
wait 20
screen close
cmd effect give @s minecraft:resistance 100000 4 true
wait 40
key dismiss
wait 10
screen close
cmd gamerule advance_time false
cmd time set 12600
cmd gamemode creative
# the study's chunks first: a busy runner's server can fall seconds behind the script (fills there fail unloaded)
cmd tp @s 0.5 201 0.5 0 20
cmd forceload add -16 -16 16 16
wait 300
# ---- a small study high over spawn: a stone floor, shelves behind the table, timber walls, a dark roof, two lanterns ----
cmd fill -6 200 -4 6 200 8 minecraft:polished_deepslate
cmd fill -6 201 8 6 204 8 minecraft:bookshelf
cmd fill -6 201 -4 -6 204 8 minecraft:spruce_planks
cmd fill 6 201 -4 6 204 8 minecraft:spruce_planks
cmd fill -6 205 -4 6 205 8 minecraft:dark_oak_planks
cmd setblock 0 201 4 awaken:inscribing_table[facing=north]
cmd setblock -2 201 5 minecraft:lantern
cmd setblock 2 201 5 minecraft:lantern
wait 40
cmd gamemode survival
cmd xp add @s 40 levels
cmd give @s minecraft:lapis_lazuli 32
cmd awaken admin enchant learn all
# one of our swords, so the written enchantment shows as its own light (SPEC §45.11)
cmd item replace entity @s weapon.mainhand with awaken:gatewrought_longsword
wait 60
screen close
# the player fell while the study was built under them: back up onto its floor
cmd tp @s 0.5 201 0.5 0 20
wait 40
# ---- live: step up and open the table, then hold for a person (or computer use) to trace with a real mouse ----
cmd tp @s 0.5 201 2.6 0 34
wait 30
use
wait 18000
quit
