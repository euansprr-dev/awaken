# Fights stage 5, the sixth Proving (SPEC §47.10): the Last Loop in the Cradle's antechamber, in a practice Proving
# (/awaken admin trial SOVEREIGN). A burst of quick shots catches the wake on the causeway looking north across the clock
# to the Seed's door; then three steps north into the Loop, a turn round, and a burst every 10 ticks catches your path
# lighting up behind you and the Last Loop kneeling where you stood, then standing. Then the Last Loop held still facing
# you, and as a spectator: the clock from the south, the Seed's door, the arch, night sweeping round at three quarters
# (/awaken admin trial 70), the outer ring falling at half (trial 45), the door turning and opening (trial 20, then 5), and
# the win: the door open, the path of light, the day coming back, the tunnel.
# Run on CI: tools/tour-ci.sh tools/autopilot/fights-antechamber.txt
world autopilot_fights_antechamber creative 20261103

# --- it faces you (held still for the shots: in creative it has no one to fight and wanders off), 9 ahead of you at
# (0, 1, 8), so the views after are measured from there ---
cmd tp @s ~ ~ ~ 0 0
wait 2
cmd data merge entity @e[type=awaken:proving_last_loop,limit=1] {NoAI:1b}
cmd tp @e[type=awaken:proving_last_loop,limit=1] ^ ^ ^9 facing entity @p

# --- the clock from high in the south (0, 14, 20), looking north to the Seed's door ---

# --- the Seed's door from the floor (0, 3, -12) ---
cmd tp @s ~ ~-11 ~-33 180 -16

# --- the arch and the causeway from (0, 3, 5), looking south ---
cmd tp @s ~ ~ ~18 0 -4

# --- at three quarters night sweeps round the clock, from high (0, 15, 9) looking north ---
cmd tp @s ~ ~12 ~4 180 45
wait 20

wait 30
shot loop_05_night_sweep
wait 50
shot loop_06_night

# --- at half the outer ring cracks and falls a mark at a time from twelve, from (0, 16, 11) ---

shot loop_08_falling_more

# --- from a quarter the Seed's door turns and opens, from the floor (0, 4, -10) ---

wait 30
shot loop_10_door_turning

shot loop_11_door_wide

# --- beaten: the door opens all the way, a path of light to it, then the day comes back; then the tunnel ---
cmd kill @e[type=awaken:proving_last_loop]

shot loop_12_won

shot loop_13_day
