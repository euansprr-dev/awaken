# Fights stage 3, the first Proving (SPEC §47.4): the Drowned Belfry and the Ringer, in a practice Proving
# (/awaken admin trial D). You wake knee-deep inside the door; then, as a spectator (never its target): the door and the
# stair out, the nave from above, the Tenor over the sealed pit, a drowned aisle, the Ringer dropped into the light when
# you stand on the chancel; then armed in front of it (creative), three blows on its stance, the stance broken
# (/awaken admin trial break) and an Execution on it; then the water draining at half its health (/awaken admin trial
# half), the floor over the pit given way, and its chain at your feet. Every spot until the drop is off the chancel, so
# it drops on cue.
# Run on CI: tools/tour-ci.sh tools/autopilot/fights-belfry.txt
world autopilot_fights_belfry creative 20261102
waitworld
cmd gamerule send_command_feedback false
cmd gamerule awaken:fauna_spawning false
cmd gamerule spawn_monsters false
cmd gamerule advance_time false
cmd weather clear 100000
cmd time set 6000
cmd awaken opening skip
wait 100
screen close
cmd awaken powers starter 1
wait 20
screen close
cmd awaken admin trial D
wait 200
hud off

# --- you wake inside the door, facing the chancel (0, 0, 48 from the hall's centre) ---
look 180 0
wait 20
shot belfry_01_wake

# --- the door and the stair out, from the narthex (0, 2, 36) ---
cmd gamemode spectator
cmd tp @s ~ ~2 ~-12 0 5
wait 30
shot belfry_02_door

# --- the nave from above the narthex (0, 22, 42) ---
cmd tp @s ~ ~20 ~6 180 30
wait 30
shot belfry_03_nave

# --- the Tenor over the sealed pit (0, 8, -14) ---
cmd tp @s ~ ~-14 ~-56 180 -12
wait 30
shot belfry_04_tenor

# --- a drowned aisle (24, 2, -2) ---
cmd tp @s ~24 ~-6 ~12 180 8
wait 30
shot belfry_05_aisle

# --- onto the chancel (0, 2, 6): it drops into the light ---
cmd tp @s ~-24 ~ ~8 180 0
wait 60
shot belfry_06_ringer
cmd summon minecraft:marker ~ ~ ~ {Tags:["belfry_chancel"]}

# --- you, armed, in front of it (creative, so it never turns on you), looking straight back along its facing: three
# blows land braced, and the stance under its health shows them (SPEC §47.3, "Seen"). A shot stalls the client while
# the server ticks on, so each is taken with the server frozen (the broken window is 40 server ticks). Arms starts no
# move in flight, and creative keeps a spectator's flying: a tick of survival puts you on your feet first ---
cmd gamemode survival
cmd gamemode creative
cmd item replace entity @s weapon.mainhand with awaken:gatewrought_longsword
cmd execute at @e[type=awaken:proving_ringer,limit=1] rotated as @e[type=awaken:proving_ringer,limit=1] rotated ~ 0 run tp @s ^ ^ ^3 ~180 3
wait 20
hud on
arms strike
wait 12
arms strike
wait 12
arms strike
wait 10
cmd tick freeze
shot belfry_06b_stance
cmd tick unfreeze

# --- its stance broken (/awaken admin trial break): down on a knee, and the plate says how to finish it ---
cmd awaken admin trial break
wait 6
cmd tick freeze
shot belfry_06c_broken
cmd tick unfreeze

# --- a heavy held 12 ticks and let go on it: the Execution, and the 7-10% of its health it takes ---
arms heavy 12
wait 15
cmd tick freeze
shotnow belfry_06d_execution
cmd tick unfreeze
wait 30
shot belfry_06e_executed
hud off
cmd gamemode spectator
cmd execute at @e[type=minecraft:marker,tag=belfry_chancel,limit=1] run tp @s ~ ~ ~ 180 0
wait 10

# --- at half its health the water runs off north toward the pit ---
cmd awaken admin trial half
wait 10
shot belfry_07_drain
wait 60

# --- the floor over the pit has given way (0, 10, -22) ---
cmd tp @s ~ ~8 ~-28 180 40
wait 20
shot belfry_08_pit

# --- beaten: its chain at your feet (0, 1, -18) ---
cmd tp @s ~ ~-9 ~4 180 55
wait 10
cmd kill @e[type=awaken:proving_ringer]
wait 60
hud on
shot belfry_09_chain
hud off
wait 40
look 180 -20
shot belfry_10_after
hud on
quit
