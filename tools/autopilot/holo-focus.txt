# Lens focus behind an open holo panel (Lens texel 14, UiHooks.focus): the same view with the K menu closed and open,
# with Lens Swift and with Lens off, by day, at night and in a dark room, so the dimming shows and the panel stays bright.
# Run: tools/autopilot.sh tools/autopilot/holo-focus.txt
world autopilot_holo_focus creative 1234
waitworld
cmd gamerule send_command_feedback false
cmd time set 6000
cmd weather clear
look 20 4
gui 0
cmd awaken admin awaken
wait 200
cmd awaken powers starter 1
wait 20
screen close
cmd fill ~-12 ~ ~-12 ~12 ~3 ~12 air replace minecraft:short_grass
cmd fill ~-12 ~ ~-12 ~12 ~3 ~12 air replace minecraft:tall_grass
lens swift
wait 40

# --- day: closed, then open (Lens Swift), then open with Lens off ---
shot focus_01_day_closed
screen holo
wait 40
shot focus_02_day_open
lens off
wait 20
shot focus_03_day_open_lens_off
lens swift
wait 20
look 20 -30
wait 15
shot focus_04_sky_open
look 20 4
screen close
wait 40
shot focus_05_day_closed_after

# --- night ---
cmd time set 18000
wait 30
screen holo
wait 40
shot focus_06_night_open
screen close
wait 40
shot focus_07_night_closed
cmd time set 6000
wait 20
quit
