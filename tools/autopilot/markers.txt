# blocks, at noon (nothing), dusk (the shaft at its brightest) and midnight (faint shaft, glowing pillars).

waitworld
gui 0

cmd gamerule awaken:fauna_spawning false

wait 40

cmd time set 6000
wait 30
shot mk_vault_48_noon
cmd time set 12400
wait 40
shot mk_vault_48_dusk
cmd tp @s ~ ~ ~72
wait 40
shot mk_vault_120_dusk
cmd tp @s ~ ~ ~100
wait 40

wait 40

wait 40
