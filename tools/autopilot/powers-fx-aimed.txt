# Aimed powers in first person (how the caster sees a beam), husk targets 4.5 ahead. Run by the integrator.
world powerfx creative 4242
waitworld
gui 0
cmd gamerule awaken:fauna_spawning false
cmd gamerule send_command_feedback false
cmd gamerule spawn_mobs false
cmd weather clear 100000
cmd time set 6000
cmd awaken admin awaken
wait 200
cmd awaken powers starter 1
wait 20
screen close
cmd tp @s 0.5 200 0.5 0 8
cmd fill -12 199 -8 12 199 22 minecraft:smooth_stone
cmd fill -12 200 -8 12 216 22 minecraft:air
wait 40

# beam
cmd fill -12 200 -8 12 216 22 minecraft:air
cmd fill -12 199 -8 12 199 22 minecraft:smooth_stone

cmd summon minecraft:husk 0.5 200 4.5 {NoAI:1b,PersistenceRequired:1b,Rotation:[180f,0f]}
cmd awaken admin power reset
wait 8
look 0 8
cmd awaken admin power cast beam
shot fp_beam_1
wait 2
shot fp_beam_2
shot fp_beam_3
shot fp_beam_4
