# The First Vista (docs/plan-pack.html, stage 2): a fresh world seen from above and from the ground once far terrain
# has been built (Distant Horizons pregen), with Lens. Then a content check: Awaken's ruins and creatures must still
# place in this terrain. Run with PACK=perf,horizon,world, DH=swift|cinematic and LENS=off|swift|cinematic; the same
# seed every time, so runs with different settings give matching shots.
world vista creative 20261024
waitworld
gui 0
cmd weather clear 1000000
cmd time set 6000
cmd awaken admin awaken
wait 200
cmd awaken powers starter 1
wait 20
screen close
window 1440 900
info
# stand on dry land (the seed's spawn is a deep ocean), then build far terrain around it before any shot
cmd awaken gate hop
wait 200
cmd dh pregen start minecraft:overworld ~ ~ 64
wait 6000
cmd dh pregen status
look 0 -4
wait 60
shot vista_ground_south
look 180 -4
wait 60
shot vista_ground_north
perf vista_ground 10
cmd gamemode spectator
cmd tp @s ~ ~90 ~
wait 400
perf vista_noon 10
look 0 16
wait 60
shot vista_noon_south
look 90 16
wait 60
shot vista_noon_west
look 180 16
wait 60
shot vista_noon_north
look -90 16
wait 60
shot vista_noon_east
cmd time set 12350
wait 120
look -90 12
wait 60
shot vista_sunset_east
look 90 12
wait 60
shot vista_sunset_west
perf vista_sunset 10
# content check: the System must still find Awaken's places and creatures in this terrain
cmd locate structure awaken:precursor_vault
wait 100
cmd locate structure awaken:precursor_monolith
wait 100
cmd locate structure awaken:precursor_ocean_lab
wait 100
cmd locate structure awaken:precursor_sky_platform
wait 100
cmd gamemode creative
cmd awaken fauna census
wait 40
quit
