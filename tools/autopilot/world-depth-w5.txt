# Between the Gates W5, "Caves worth entering" (SPEC §28.5): one look tour on GitHub's Mac runners, at the places this
# world really made (`awaken delve goto <kind>` finds the nearest with each candidate's own site search and stands you at
# its way in; with none within 1,600 it lays one). A sinkhole from the rim in the morning (its mist, the caption), across
# at its shelves, from the diving lip, on a shelf, its vine cleft, its grotto's lode; the Delvers' mine (the adit, the
# office, the drowned stope's cut walls, the minecart at the incline, the bridge); a chute from its mouth and from the
# lake; a seep's violet hall, its heart asleep, awake and dying; a grotto's room, its geode and its curtain; the Crystal
# Nest's tunnel out; a vent; the way home's thread; a rappel (the Ascent's pilot shoots its frames); the cave biomes' looks.
# Run: tools/tour-ci.sh tools/autopilot/world-depth-w5.txt 1800 LENS=cinematic
world autopilot_world_depth_w5 creative 52605260
waitworld
gui 0
cmd gamerule send_command_feedback false
cmd gamerule spawn_mobs false
cmd gamerule awaken:fauna_spawning false
cmd weather clear 1000000
cmd awaken opening skip
wait 100
screen close
cmd awaken powers starter 1
wait 20
screen close
wait 40
key dismiss
wait 10
screen close
# the director holds paced lines for a world's first five minutes: run them through
cmd tick sprint 6200
wait 240
cmd gamerule advance_time false
cmd time set 1000
cmd effect give @s minecraft:resistance 100000 4 true
lens cinematic
camera first
# ---- the Horizon line at spawn: the first sinkhole, "The Opening · a way down" ----
hud on
wait 60
shot w5_00_horizon_at_spawn

# ---- a sinkhole: from the rim (its mist, its caption), its shelves, the lip, a shelf, the cleft, its grotto ----
# goto keeps your pitch: look down into the hole from the rim (from eye height a level gaze sees only its far lip),
# and shoot the caption while it shows (it holds about five seconds)
look 0 32
cmd awaken delve goto sinkhole
wait 70
shotnow w5_01_sinkhole_caption
wait 80
hud off
shot w5_02_sinkhole_rim_mist
cmd awaken delve place go view shelf
wait 40
shot w5_03_sinkhole_shelves
cmd awaken delve place go dive pool
wait 30
shot w5_04_sinkhole_dive_lip
cmd awaken delve place go shelf pool
wait 30
shot w5_05_sinkhole_on_a_shelf
cmd awaken delve place go cleft view
wait 30
shot w5_06_sinkhole_vine_cleft_from_floor
cmd awaken delve place go goal heart
wait 40
shot w5_07_sinkhole_grotto_lode
